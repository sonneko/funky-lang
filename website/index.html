About Funky

          

            
Funky is a functional-first, expression-oriented, and statically typed language.

            
Key Features

            
    
              
  • Structural Typing: Support for structs and algebraic data types (enums).
    • 
              
  • Protocol-based Polymorphism: Define and use protocols (traits/interfaces) with ^ prefix.
    • 
              
  • Type Generics and Constraints: Clearly define what can be called.
    • 
              
  • Curried Functions: Functions are curried by default.
    • 
              
  • Do-Where Blocks: Sequencing expressions and defining local variable bindings.
    • 
              
  • Type Prefixes: Types and generic parameters use $ prefix.
    •

Syntax Examples

            

$Point is { x = $Int y = $Int };

add x: $Int y: $Int > $Int is
  x + y;

$Option is $T => | Some = $T None = $Unit |;

main > $Unit is
  do {
    log "Hello, Funky!";
  } where {
    log = #console_log
  };
            

            
Grammar

            

<program> ::= <top_level>*
<top_level> ::= <using> | <function> | <type_definition> | <protocol_definition>

# using
<using> ::= "using" ( <identifier> ( "as" <identifier> )? )* "from" <literal> ";"

# function
<function> ::= <identifier> <protocol_def_list>? <param_def_list> "is"
     ( <expression> | "#" <identifier> ) ";"
<protocol_def_list> ::= ( "$" <identifier>: <protocol_definition_body> )* "=>"
<param_def_list> ::= <param_def>* ">" <type_literal>
<param_def> ::= <identifier> ":" <type_literal>

<expression> ::= 
     | <if_expression>
     | <call_expression>
     | <block_expression>
     | <primary_expression>
     | <paren_expression>

<if_expression> ::= "if" <expression> "then" <expression> "else" <expression>
<call_expression> ::= <expression> <primary_expression>
<block_expression> ::= "do" "{"
         ( <expression> ";" | "break" <expression> )*
         <expression>
     "}" "where" "{"
         ( <identifier> ( ":" <type_literal> ) "=" <expression> )*
     "}"
<primary_expression> ::= <literal> | <struct_literal> | <enum_literal> | <period_access>
<struct_literal> ::= <identifier> "{"
     ( <identifier> "=" <expression> )*
"}"
<enum_literal> ::= <identifier> "(" <expression> ")"
<paren_expression> ::= "(" <expression> ")"
<period_access> ::= <identifier> ( "." <identifier> )*

# types
<type_definition> ::= "$" <identifier> "is" <type_literal> ";"
<type_literal> ::=
     | <struct_type_body>
     | <enum_type_body>
     | <fn_type_body>
     | "$" <identifier> ( "<" <type_literal>* ">" )?
<struct_type_body> ::= (( "$" <identifier> )* "=>" )? "{" (<identifier> "=" <type_literal>)* "}"
<enum_type_body> ::= (( "$" <identifier> )* "=>" )? "|" ( <identifier> "=" <type_literal> )* "|"
<fn_type_body> ::= ( ( "$" <identifier> ":" <protocol_literal> )* "=>" )? <type_literal> ">" ( <fn_type_body> | <type_literal> )

# protocols
<protocol_definition> ::= "^" <identifier> "is" <protocol_definition_body> ";"
<protocol_definition_body> ::= "{"
      ( <identifier> "=" <fn_type_body> )*
"}" | <protocol_literal>
<protocol_literal> ::= "^" <identifier> ( "+" "^" <identifier> )*
